Google Sheets 자동 연동 · GitHub Pages 배포 · 직원 휴대폰 공유
📌 한 줄 요약 이 가이드를 따라하면 직원이 자기 휴대폰에서 URL 한 번 열어 서명만 하면 모든 기록이 회사 구글 스프레드시트 한 곳에 자동으로 쌓입니다. 최초 1회 약 15분 설정.

3 💾 저장(Ctrl/⌘+S)

/**
 * 한양전공(주) 법정의무교육 전자서명 백엔드
 * - POST: 서명 1건을 '서명기록' 시트에 추가 (날짜+교육+사람 기준 일별 중복 차단)
 * - GET ?action=list: 시트의 모든 서명을 JSON으로 반환
 *
 * 중복 차단 규칙:
 *   같은 사람이 같은 교육에 같은 날짜 → 차단
 *   같은 사람이 같은 교육에 다른 날짜 → 허용 (분기/연간 반복 교육 대비)
 *   같은 사람이 다른 교육에 같은 날짜 → 허용
 */
const HEADER = ['일시','날짜','교육명','부서','직급','이름','서명이미지','단말정보'];
const IMG_FOLDER = '한양전공_서명이미지';

function _sheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName('서명기록');
  if (!sh) {
    sh = ss.insertSheet('서명기록');
    sh.appendRow(HEADER);
    sh.getRange('A1:H1').setFontWeight('bold').setBackground('#1f6b3a').setFontColor('#fff');
    sh.setFrozenRows(1);
    sh.setColumnWidth(1, 150);  // 일시
    sh.setColumnWidth(2, 100);  // 날짜
    sh.setColumnWidth(3, 220);  // 교육명
    sh.setColumnWidth(7, 220);  // 서명이미지
  }
  return sh;
}

function _folder() {
  const it = DriveApp.getFoldersByName(IMG_FOLDER);
  return it.hasNext() ? it.next() : DriveApp.createFolder(IMG_FOLDER);
}

function _json(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const sh = _sheet();
    const data = JSON.parse(e.postData.contents);
    const atDate = data.atDate || Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd');

    // 서버 측 중복 차단: 같은 날짜 + 같은 교육 + 같은 사람
    const all = sh.getDataRange().getValues();
    for (let i = 1; i < all.length; i++) {
      const rowDate = typeof all[i][1] === 'string'
        ? all[i][1]
        : Utilities.formatDate(new Date(all[i][1]), 'Asia/Seoul', 'yyyy-MM-dd');
      if (rowDate === atDate &&
          all[i][2] === data.trainingName &&
          all[i][3] === data.dept &&
          all[i][5] === data.name) {
        return _json({
          ok: false, error: 'duplicate',
          message: '같은 날짜에 이미 동일 교육 서명이 존재합니다.'
        });
      }
    }

    // 서명 이미지를 드라이브에 저장 + 시트에 IMAGE 미리보기
    let imageFormula = '';
    if (data.img && data.img.startsWith('data:image/')) {
      const blob = Utilities.newBlob(
        Utilities.base64Decode(data.img.split(',')[1]),
        'image/png',
        `${atDate}_${data.dept}_${data.name}_${new Date().getTime()}.png`
      );
      const file = _folder().createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      imageFormula = `=IMAGE("https://drive.google.com/uc?export=view&id=${file.getId()}", 4, 30, 90)`;
    }

    sh.appendRow([
      data.atStr || new Date().toLocaleString('ko-KR'),
      atDate,
      data.trainingName || '',
      data.dept || '',
      data.rank || '',
      data.name || '',
      '',
      data.userAgent || ''
    ]);
    if (imageFormula) {
      const lr = sh.getLastRow();
      sh.getRange(lr, 7).setFormula(imageFormula);
      sh.setRowHeight(lr, 50);
    }
    return _json({ ok: true, row: sh.getLastRow() });
  } catch (err) {
    return _json({ ok: false, error: err.toString() });
  }
}

function doGet(e) {
  try {
    if (e && e.parameter && e.parameter.action === 'list') {
      const sh = _sheet();
      const all = sh.getDataRange().getValues();
      const numRows = all.length - 1;
      const imgFormulas = numRows > 0 ? sh.getRange(2, 7, numRows, 1).getFormulas() : [];
      const records = [];
      for (let i = 1; i < all.length; i++) {
        const row = all[i];
        if (!row[2] || !row[5]) continue;
        const atDate = typeof row[1] === 'string'
          ? row[1]
          : Utilities.formatDate(new Date(row[1]), 'Asia/Seoul', 'yyyy-MM-dd');
        const imgFormula = imgFormulas[i-1] ? imgFormulas[i-1][0] : '';
        const idMatch = imgFormula.match(/id=([^"&,]+)/);
        const imgUrl = idMatch ? 'https://drive.google.com/thumbnail?id=' + idMatch[1] + '&sz=w300' : '';
        records.push({
          atStr: typeof row[0] === 'string' ? row[0] : new Date(row[0]).toLocaleString('ko-KR'),
          atDate: atDate,
          trainingName: row[2],
          dept: row[3],
          rank: row[4],
          name: row[5],
          img: imgUrl
        });
      }
      return _json({ ok: true, records });
    }
    return ContentService
      .createTextOutput('한양전공(주) 전자서명 API 정상 작동 · ' + new Date().toLocaleString('ko-KR'))
      .setMimeType(ContentService.MimeType.TEXT);
  } catch (err) {
    return _json({ ok: false, error: err.toString() });
  }
}

3단계 — 웹앱으로 배포

1 우측 상단 배포 → 새 배포 클릭

2 유형 선택 옆 ⚙ 톱니바퀴 → 웹 앱 선택

3 설정:

  • 설명: 한양전공 전자서명 v1
  • 다음 사용자로 실행: 나
  • 액세스 권한이 있는 사용자: 모든 사용자

4 배포 → 권한 승인 요청 시 액세스 승인 → 본인 계정 → 고급 → "(안전하지 않음) 이동" → 허용

5 표시된 웹 앱 URL 복사 (https://script.google.com/macros/s/...../exec)

⚠ "안전하지 않음" 경고가 뜨는 이유 구글이 검증하지 않은 본인 작성 스크립트이기 때문이며, 본인 회사 시트에만 쓰는 코드이므로 안전합니다. "고급 → 이동 → 허용"으로 진행하시면 됩니다.

4단계 — GitHub Pages에 그대로 업로드

1 github.com 로그인 → 본인 저장소(예: edu-sign)로 이동

2 Add file → Upload files 클릭 → index.html과 guide.html을 편집 없이 그대로 끌어다 놓기

3 화면 하단 Commit changes 클릭

4 Settings → Pages 메뉴에서 Pages가 켜져 있으면 1~2분 뒤 자동 반영됩니다.

5단계 — 최초 설정 & 직원 공유

1 https://[사용자명].github.io/edu-sign/ 주소를 본인 PC나 휴대폰에서 엽니다.

2 "⏳ 시스템 준비 중입니다" 화면이 뜨면 하단의 관리자 설정 버튼 클릭 → 비밀번호 123456 입력

3 🚀 최초 1회 설정 화면에서 3단계에서 복사한 Apps Script URL을 붙여넣고 공유 URL 생성 클릭

4 생성된 공유 URL(...github.io/edu-sign/?gs=...)을 📋 URL 복사하기 버튼으로 복사

5 복사한 URL을 카톡 단체방·문자·이메일로 직원들에게 공유. 관리자 본인도 이 URL을 북마크해 두세요.

✅ 잘 설정됐는지 확인하는 법 생성된 공유 URL을 열어 우측 상단 관리자 → 123456을 입력하면 상단에 "📡 Google Sheets 연결됨" 초록 카드가 떠야 합니다. 빨간 카드(✕ 연결 실패)가 뜨면 Apps Script URL이 잘못되었거나 권한 설정 문제이므로 3단계를 다시 확인해 주세요.

운영 흐름

1) 교육 진행 후 카톡으로 URL 전송 → "각자 휴대폰에서 30초 안에 서명해주세요" 안내

2) 직원: URL 접속 → 교육 선택 → 부서/이름 선택 → 손가락으로 서명 → 제출

3) 관리자: 본인 휴대폰/PC에서 같은 URL 열고 관리자 진입 → 실시간으로 서명 현황 확인 → 미서명자 명단 복사해서 미참석자에게 푸시

4) 스프레드시트 확인: 구글 시트에 직접 들어가면 일시/교육명/부서/직급/이름과 함께 서명 이미지 미리보기까지 한 줄씩 쌓여 있습니다. 그대로 PDF로 출력하여 법정의무교육 증빙으로 제출하시면 됩니다.

주의사항

🔒 관리자 비밀번호 기본값 123456은 HTML 소스에 노출됩니다. 외부 공유 환경에서는 index.html을 열어 const ADMIN_PW = '123456'; 부분을 회사 내부에서만 아는 값으로 바꾸시기 바랍니다.
🔗 Apps Script URL 보안 GS_URL은 GitHub에 공개됩니다. 누구나 POST해서 가짜 서명을 넣을 수 있다는 의미입니다. 71명 규모 사내 운영에서는 보통 문제 없지만, 가짜 서명이 발견되면 관리자가 시트에서 직접 행을 삭제하면 됩니다.

문제 해결

  • 관리자 화면에 "연결 실패" — index.html의 GS_URL이 따옴표 포함해서 정확히 들어갔는지, URL 끝이 /exec인지 확인 (끝이 /dev면 안 됨)
  • 중복 서명이 안 막혀요 — 서명 후 시트에 데이터가 들어왔는지 먼저 확인. 시트에 들어왔다면 다른 폰에서 다시 열었을 때 첫 로딩이 끝나고 중복 차단이 작동합니다 (약 2초).
  • 이미지가 #REF! 로 표시돼요 — 시트 새로고침 또는 IMAGE 함수 재계산.
  • 코드를 수정했는데 적용이 안 됨 — Apps Script에서 배포 → 배포 관리 → 연필(편집) → 버전 "새 버전"으로 재배포 필요.
